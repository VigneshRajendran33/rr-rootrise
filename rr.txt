# RR Devi Grocery Store

Grocery delivery PWA for Mettur, Tamil Nadu — React + Vite + Firebase + Razorpay.

## Stack
React 19, Vite, Firebase (Auth/Firestore/Storage/FCM/Functions), Razorpay, Google Maps, Framer Motion, Lucide React.

## Setup

npm install
cp .env.example .env.local
npm run dev


## Cloud Functions

cd functions
npm install
firebase functions:secrets:set RAZORPAY_KEY_ID
firebase functions:secrets:set RAZORPAY_KEY_SECRET
firebase deploy --only functions


## Firestore & Storage rules

firebase deploy --only firestore:rules
firebase deploy --only storage


## Build & Deploy (Vercel)

npm run build
vercel --prod


## Android (Capacitor)

npm run build
npx cap add android
npx cap sync android
npx cap open android


## Admin access
Admin accounts are created manually in Firestore's `admins/{uid}` collection
(matching a real Firebase Auth email/password account) — there is no public
admin signup.

## Folder structure
- `src/pages/` — route-level pages (customer, `admin/`, `delivery/`)
- `src/components/` — shared UI, organized by domain
- `src/context/` — Auth, Cart, Wishlist, Language, Admin/Delivery role guards
- `src/firebase/services/` — Firestore data access functions
- `functions/` — Cloud Functions (payments, notifications, rate limiting)

## Known incomplete areas
- Admin CMS (Part 3 spec) — only a scoped Orders/payment view exists, not the
  full product/category/banner/offer/coupon management dashboard.
- Wishlist is local-state only, not yet synced to Firestore.
- Delivery History and Delivery Profile pages are not built.
- Reviews collection exists with rules but has no submission/read UI yet.
- `/wishlist` route is linked from the header but has no page behind it.

const { onCall, HttpsError } = require('firebase-functions/v2/https');
const { onDocumentCreated, onDocumentUpdated } = require('firebase-functions/v2/firestore');
const { defineSecret } = require('firebase-functions/params');
const admin = require('firebase-admin');
const Razorpay = require('razorpay');
const crypto = require('crypto');

admin.initializeApp();
const db = admin.firestore();

const razorpayKeyId = defineSecret('RAZORPAY_KEY_ID');
const razorpayKeySecret = defineSecret('RAZORPAY_KEY_SECRET');

const DELIVERY_FEE = 25;
const FREE_DELIVERY_THRESHOLD = 499;
const TAX_RATE = 0.05;
const RATE_LIMIT_WINDOW_MS = 60 * 1000;
const RATE_LIMIT_MAX_REQUESTS = 5;

async function checkRateLimit(uid, action) {
  const rateLimitRef = db.collection('rateLimits').doc(`${uid}_${action}`);
  const snap = await rateLimitRef.get();
  const now = Date.now();

  if (!snap.exists) { await rateLimitRef.set({ count: 1, windowStart: now }); return; }

  const data = snap.data();
  if (now - data.windowStart > RATE_LIMIT_WINDOW_MS) {
    await rateLimitRef.set({ count: 1, windowStart: now });
    return;
  }
  if (data.count >= RATE_LIMIT_MAX_REQUESTS) {
    throw new HttpsError('resource-exhausted', 'Too many attempts. Please wait a moment and try again.');
  }
  await rateLimitRef.update({ count: admin.firestore.FieldValue.increment(1) });
}

async function getUserContactInfo(uid) {
  const userSnap = await db.collection('users').doc(uid).get();
  if (!userSnap.exists) return { customerName: '', customerPhone: '' };
  const userData = userSnap.data();
  return { customerName: userData.name ?? '', customerPhone: userData.phone ?? '' };
}

async function calculateTrustedTotal(items) {
  const productRefs = items.map((item) => db.collection('products').doc(item.id));
  const snaps = await db.getAll(...productRefs);
  let subtotal = 0;
  const validatedItems = [];

  snaps.forEach((snap, i) => {
    if (!snap.exists) throw new HttpsError('failed-precondition', 'An item in your cart is no longer available.');
    const product = snap.data();
    if (product.status !== 'active') throw new HttpsError('failed-precondition', `${product.name} is currently unavailable.`);
    if (product.stock < items[i].qty) throw new HttpsError('failed-precondition', `${product.name} doesn't have enough stock.`);
    subtotal += product.price * items[i].qty;
    validatedItems.push({ id: snap.id, name: product.name, qty: items[i].qty, price: product.price, thumbnailUrl: product.thumbnailUrl ?? null });
  });

  const deliveryCharge = subtotal >= FREE_DELIVERY_THRESHOLD ? 0 : DELIVERY_FEE;
  const tax = Math.round(subtotal * TAX_RATE);
  const grandTotal = subtotal + deliveryCharge + tax;
  return { subtotal, deliveryCharge, tax, grandTotal, validatedItems };
}

exports.createRazorpayOrder = onCall({ secrets: [razorpayKeyId, razorpayKeySecret] }, async (request) => {
  if (!request.auth) throw new HttpsError('unauthenticated', 'You must be signed in to check out.');
  await checkRateLimit(request.auth.uid, 'checkout');

  const { checkoutSessionId, items, addressId, deliverySlot } = request.data;
  const uid = request.auth.uid;

  if (!checkoutSessionId || !Array.isArray(items) || items.length === 0 || !addressId) {
    throw new HttpsError('invalid-argument', 'Missing required checkout information.');
  }

  const existingQuery = await db.collection('orders').where('checkoutSessionId', '==', checkoutSessionId).where('uid', '==', uid).limit(1).get();

  if (!existingQuery.empty) {
    const existingData = existingQuery.docs[0].data();
    if (existingData.paymentStatus === 'PAID') throw new HttpsError('already-exists', 'This order has already been paid for.');
    if (existingData.razorpayOrderId) {
      return { internalOrderId: existingQuery.docs[0].id, razorpayOrderId: existingData.razorpayOrderId, amount: existingData.paymentAmount, keyId: razorpayKeyId.value() };
    }
  }

  const { subtotal, deliveryCharge, tax, grandTotal, validatedItems } = await calculateTrustedTotal(items);
  const { customerName, customerPhone } = await getUserContactInfo(uid);

  const razorpay = new Razorpay({ key_id: razorpayKeyId.value(), key_secret: razorpayKeySecret.value() });
  const razorpayOrder = await razorpay.orders.create({
    amount: Math.round(grandTotal * 100), currency: 'INR', receipt: `RD-${Date.now()}`, notes: { uid, checkoutSessionId }
  });

  const orderRef = existingQuery.empty ? db.collection('orders').doc() : existingQuery.docs[0].ref;

  await orderRef.set({
    uid, customerName, customerPhone, checkoutSessionId, addressId, deliverySlot,
    items: validatedItems, subtotal, deliveryCharge, tax, grandTotal,
    paymentMethod: 'RAZORPAY', paymentStatus: 'PENDING', orderStatus: 'PLACED',
    deliveryStatus: null, deliveryBoyId: null,
    razorpayOrderId: razorpayOrder.id, paymentAmount: grandTotal, currency: 'INR',
    paymentCreatedAt: admin.firestore.FieldValue.serverTimestamp(), status: 'active',
    createdAt: admin.firestore.FieldValue.serverTimestamp(), updatedAt: admin.firestore.FieldValue.serverTimestamp()
  }, { merge: true });

  return { internalOrderId: orderRef.id, razorpayOrderId: razorpayOrder.id, amount: grandTotal, keyId: razorpayKeyId.value() };
});

exports.verifyRazorpayPayment = onCall({ secrets: [razorpayKeySecret] }, async (request) => {
  if (!request.auth) throw new HttpsError('unauthenticated', 'You must be signed in.');

  const { internalOrderId, razorpay_order_id, razorpay_payment_id, razorpay_signature } = request.data;
  const orderRef = db.collection('orders').doc(internalOrderId);
  const orderSnap = await orderRef.get();
  if (!orderSnap.exists) throw new HttpsError('not-found', 'Order not found.');
  const order = orderSnap.data();

  if (order.uid !== request.auth.uid) throw new HttpsError('permission-denied', 'This order does not belong to you.');
  if (order.paymentStatus === 'PAID') return { verified: true, orderId: internalOrderId };

  const expectedSignature = crypto.createHmac('sha256', razorpayKeySecret.value()).update(`${razorpay_order_id}|${razorpay_payment_id}`).digest('hex');

  if (expectedSignature !== razorpay_signature) {
    await orderRef.update({ paymentStatus: 'FAILED', paymentFailureReason: 'Signature verification failed', updatedAt: admin.firestore.FieldValue.serverTimestamp() });
    throw new HttpsError('invalid-argument', 'Payment verification failed.');
  }

  try {
    await db.runTransaction(async (transaction) => {
      const productRefs = order.items.map((item) => db.collection('products').doc(item.id));
      const productSnaps = await Promise.all(productRefs.map((ref) => transaction.get(ref)));
      productSnaps.forEach((snap, i) => {
        if (!snap.exists || snap.data().stock < order.items[i].qty) throw new Error(`STOCK_ISSUE:${order.items[i].name}`);
      });
      productSnaps.forEach((snap, i) => {
        transaction.update(productRefs[i], { stock: snap.data().stock - order.items[i].qty, updatedAt: admin.firestore.FieldValue.serverTimestamp() });
      });
      transaction.update(orderRef, {
        paymentStatus: 'PAID', razorpayPaymentId: razorpay_payment_id, razorpaySignature: razorpay_signature,
        paymentVerifiedAt: admin.firestore.FieldValue.serverTimestamp(), orderStatus: 'CONFIRMED', updatedAt: admin.firestore.FieldValue.serverTimestamp()
      });
    });
  } catch (err) {
    await orderRef.update({
      paymentStatus: 'PAID', razorpayPaymentId: razorpay_payment_id, razorpaySignature: razorpay_signature,
      paymentVerifiedAt: admin.firestore.FieldValue.serverTimestamp(), orderStatus: 'NEEDS_REVIEW',
      needsReview: true, reviewReason: err.message, updatedAt: admin.firestore.FieldValue.serverTimestamp()
    });
    return { verified: true, orderId: internalOrderId, needsReview: true };
  }

  return { verified: true, orderId: internalOrderId };
});

exports.placeCodOrder = onCall(async (request) => {
  if (!request.auth) throw new HttpsError('unauthenticated', 'You must be signed in to check out.');
  await checkRateLimit(request.auth.uid, 'checkout');

  const { checkoutSessionId, items, addressId, deliverySlot } = request.data;
  const uid = request.auth.uid;

  if (!checkoutSessionId || !Array.isArray(items) || items.length === 0 || !addressId) {
    throw new HttpsError('invalid-argument', 'Missing required checkout information.');
  }

  const existingQuery = await db.collection('orders').where('checkoutSessionId', '==', checkoutSessionId).where('uid', '==', uid).limit(1).get();
  if (!existingQuery.empty) return { orderId: existingQuery.docs[0].id, alreadyExists: true };

  const { customerName, customerPhone } = await getUserContactInfo(uid);
  const orderRef = db.collection('orders').doc();

  await db.runTransaction(async (transaction) => {
    const productRefs = items.map((item) => db.collection('products').doc(item.id));
    const productSnaps = await Promise.all(productRefs.map((ref) => transaction.get(ref)));
    let subtotal = 0;
    const validatedItems = [];

    productSnaps.forEach((snap, i) => {
      if (!snap.exists) throw new Error('Product no longer available.');
      const product = snap.data();
      if (product.status !== 'active') throw new Error(`${product.name} is currently unavailable.`);
      if (product.stock < items[i].qty) throw new Error(`${product.name} doesn't have enough stock.`);
      subtotal += product.price * items[i].qty;
      validatedItems.push({ id: snap.id, name: product.name, qty: items[i].qty, price: product.price, thumbnailUrl: product.thumbnailUrl ?? null });
    });

    const deliveryCharge = subtotal >= FREE_DELIVERY_THRESHOLD ? 0 : DELIVERY_FEE;
    const tax = Math.round(subtotal * TAX_RATE);
    const grandTotal = subtotal + deliveryCharge + tax;

    productSnaps.forEach((snap, i) => {
      transaction.update(productRefs[i], { stock: snap.data().stock - items[i].qty, updatedAt: admin.firestore.FieldValue.serverTimestamp() });
    });

    transaction.set(orderRef, {
      uid, customerName, customerPhone, checkoutSessionId, addressId, deliverySlot,
      items: validatedItems, subtotal, deliveryCharge, tax, grandTotal,
      paymentMethod: 'COD', paymentStatus: 'PENDING', orderStatus: 'PLACED',
      deliveryStatus: null, deliveryBoyId: null, status: 'active',
      orderDate: admin.firestore.FieldValue.serverTimestamp(),
      createdAt: admin.firestore.FieldValue.serverTimestamp(), updatedAt: admin.firestore.FieldValue.serverTimestamp()
    });
  });

  return { orderId: orderRef.id, alreadyExists: false };
});

async function sendPushAndLog({ uid, collectionName, title, body, type, orderId, clickAction }) {
  const targetSnap = await db.collection(collectionName).doc(uid).get();
  if (!targetSnap.exists) return;
  const tokens = targetSnap.data().fcmTokens ?? [];

  await db.collection('notifications').add({
    userId: uid, role: collectionName === 'deliveryBoys' ? 'delivery' : collectionName === 'admins' ? 'admin' : 'customer',
    title, body, type, orderId: orderId ?? null, clickAction: clickAction ?? '/', isRead: false,
    createdAt: admin.firestore.FieldValue.serverTimestamp()
  });

  if (tokens.length === 0) return;

  const response = await admin.messaging().sendEachForMulticast({
    tokens, notification: { title, body }, data: { orderId: orderId ?? '', notificationType: type, clickAction: clickAction ?? '/' }
  });

  const invalidTokens = [];
  response.responses.forEach((res, i) => {
    if (!res.success && (res.error?.code === 'messaging/registration-token-not-registered' || res.error?.code === 'messaging/invalid-registration-token')) {
      invalidTokens.push(tokens[i]);
    }
  });
  if (invalidTokens.length > 0) {
    await db.collection(collectionName).doc(uid).update({ fcmTokens: admin.firestore.FieldValue.arrayRemove(...invalidTokens) });
  }
}

import { db } from '../config';

const couponsRef = collection(db, 'coupons');

export async function validateCoupon(code, subtotal) {
  const normalizedCode = code.trim().toUpperCase();
  if (!normalizedCode) return { valid: false, reason: 'Enter a coupon code.' };

  try {
    const q = query(couponsRef, where('couponCode', '==', normalizedCode), where('status', '==', 'active'));
    const snap = await getDocs(q);

    if (snap.empty) return { valid: false, reason: 'Invalid or expired coupon.' };

    const coupon = snap.docs[0].data();

    if (coupon.expiry && new Date(coupon.expiry) < new Date()) return { valid: false, reason: 'This coupon has expired.' };
    if (coupon.usageLimit != null && coupon.usedCount >= coupon.usageLimit) return { valid: false, reason: 'This coupon has reached its usage limit.' };
    if (coupon.minimumOrder && subtotal < coupon.minimumOrder) return { valid: false, reason: `Minimum order of ₹${coupon.minimumOrder} required.` };

    const rawDiscount = coupon.discountType === 'percentage'
      ? Math.round(subtotal * (coupon.discountValue / 100))
      : coupon.discountValue;
    const discount = coupon.maximumDiscount ? Math.min(rawDiscount, coupon.maximumDiscount) : rawDiscount;

    return { valid: true, code: normalizedCode, discount, couponId: snap.docs[0].id };
  } catch (error) {
    console.error('validateCoupon failed:', error);
    return { valid: false, reason: 'Unable to validate coupon right now.' };
  }
}

import { useAuth } from './AuthContext';

const CartContext = createContext(null);
const STORAGE_KEY = 'rrdevi_cart';
const SYNC_DEBOUNCE_MS = 1200;

export function CartProvider({ children }) {
  const { user } = useAuth();
  const [items, setItems] = useState(() => {
    try {
      const saved = localStorage.getItem(STORAGE_KEY);
      return saved ? JSON.parse(saved) : [];
    } catch {
      return [];
    }
  });
  const [isDrawerOpen, setIsDrawerOpen] = useState(false);
  const hasLoadedRemoteCart = useRef(false);
  const syncTimeoutRef = useRef(null);

  useEffect(() => {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
  }, [items]);

  useEffect(() => {
    if (!user || hasLoadedRemoteCart.current) return;
    hasLoadedRemoteCart.current = true;
    getCart(user.uid).then((remoteCart) => {
      if (remoteCart?.products?.length) setItems(remoteCart.products);
    });
  }, [user]);

  useEffect(() => {
    if (!user) hasLoadedRemoteCart.current = false;
  }, [user]);

  const addItem = useCallback((product, qty = 1) => {
    setItems((prev) => {
      const existing = prev.find((i) => i.id === product.id);
      if (existing) {
        const newQty = existing.qty + qty;
        if (newQty <= 0) return prev.filter((i) => i.id !== product.id);
        return prev.map((i) => (i.id === product.id ? { ...i, qty: newQty } : i));
      }
      return qty > 0 ? [...prev, { ...product, qty }] : prev;
    });
  }, []);

  const removeItem = useCallback((productId) => {
    setItems((prev) => prev.filter((i) => i.id !== productId));
  }, []);

  const clearCart = useCallback(() => setItems([]), []);
  const openDrawer = useCallback(() => setIsDrawerOpen(true), []);
  const closeDrawer = useCallback(() => setIsDrawerOpen(false), []);

  const itemCount = items.reduce((sum, i) => sum + i.qty, 0);
  const subtotal = items.reduce((sum, i) => sum + i.price * i.qty, 0);

  useEffect(() => {
    if (!user) return;
    if (syncTimeoutRef.current) clearTimeout(syncTimeoutRef.current);
    syncTimeoutRef.current = setTimeout(() => {
      saveCart(user.uid, { products: items, subtotal, grandTotal: subtotal });
    }, SYNC_DEBOUNCE_MS);
    return () => clearTimeout(syncTimeoutRef.current);
  }, [items, user, subtotal]);

  return (
    <CartContext.Provider value={{ items, addItem, removeItem, clearCart, itemCount, subtotal, isDrawerOpen, openDrawer, closeDrawer }}>
      {children}
    </CartContext.Provider>
  );
}

javascript
import { useState, useEffect, useCallback } from 'react';
import { collection, query, where, orderBy, onSnapshot, doc, updateDoc, deleteDoc } from 'firebase/firestore';
import { db } from '../firebase/config';

export function useNotifications(uid) {
  const [notifications, setNotifications] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (!uid) { setNotifications([]); setLoading(false); return; }
    const q = query(collection(db, 'notifications'), where('userId', '==', uid), orderBy('createdAt', 'desc'));
    const unsubscribe = onSnapshot(q, (snap) => {
      setNotifications(snap.docs.map((d) => ({ id: d.id, ...d.data() })));
      setLoading(false);
    }, (err) => {
      console.error('Notifications listener failed:', err);
      setLoading(false);
    });
    return unsubscribe;
  }, [uid]);

  const markAsRead = useCallback(async (notificationId) => {
    await updateDoc(doc(db, 'notifications', notificationId), { isRead: true });
  }, []);

  const removeNotification = useCallback(async (notificationId) => {
    await deleteDoc(doc(db, 'notifications', notificationId));
  }, []);

  const unreadCount = notifications.filter((n) => !n.isRead).length;

  return { notifications, loading, unreadCount, markAsRead, removeNotification };
}

import { useState, useEffect, useCallback } from 'react';

const DISMISS_KEY = 'rrdevi_install_dismissed';

function isStandalone() {
  return window.matchMedia('(display-mode: standalone)').matches || window.navigator.standalone === true;
}

function isIOS() {
  return /iphone|ipad|ipod/i.test(window.navigator.userAgent) && !window.MSStream;
}

export function usePWAInstall() {
  const [deferredPrompt, setDeferredPrompt] = useState(null);
  const [installed, setInstalled] = useState(isStandalone());
  const [dismissed] = useState(() => localStorage.getItem(DISMISS_KEY) === 'true');
  const [isDismissedNow, setIsDismissedNow] = useState(dismissed);

  useEffect(() => {
    const handleBeforeInstall = (e) => { e.preventDefault(); setDeferredPrompt(e); };
    const handleInstalled = () => { setInstalled(true); setDeferredPrompt(null); };

    window.addEventListener('beforeinstallprompt', handleBeforeInstall);
    window.addEventListener('appinstalled', handleInstalled);
    return () => {
      window.removeEventListener('beforeinstallprompt', handleBeforeInstall);
      window.removeEventListener('appinstalled', handleInstalled);
    };
  }, []);

  const promptInstall = useCallback(async () => {
    if (!deferredPrompt) return;
    deferredPrompt.prompt();
    await deferredPrompt.userChoice;
    setDeferredPrompt(null);
  }, [deferredPrompt]);

  const dismiss = useCallback(() => {
    localStorage.setItem(DISMISS_KEY, 'true');
    setIsDismissedNow(true);
  }, []);

  const iosDevice = isIOS();
  const showBanner = !installed && !isDismissedNow && (deferredPrompt || iosDevice);

  return { showBanner, promptInstall, dismiss, isIOS: iosDevice, canPrompt: !!deferredPrompt };
}

    super(props);
    this.state = { hasError: false };
  }

    return { hasError: true };
  }

  componentDidCatch(error, info) {
    console.error('Unhandled UI error:', error, info);
  }

  handleRetry = () => this.setState({ hasError: false });
  handleGoHome = () => { window.location.href = '/'; };

  render() {
    if (this.state.hasError) {
      return (
        <div className="rrdevi-error-boundary">
          <p>Something went wrong.</p>

            <button onClick={this.handleRetry}><RefreshCw size={14} /> Retry</button>
            <button onClick={this.handleGoHome}><Home size={14} /> Go Home</button>
          </div>
        </div>
      );
    }
    return this.props.children;
  }
}

import { motion } from 'framer-motion';

import { useNavigate } from 'react-router-dom';
import { useCart } from '../../context/CartContext';
import { useWishlist } from '../../context/WishlistContext';

export default function ProductCard({ product }) {
  const navigate = useNavigate();
  const { addItem, items, removeItem } = useCart();
  const { ids: wishlistIds, toggle: toggleWishlist } = useWishlist();

  const cartEntry = items.find((i) => i.id === product.id);
  const inWishlist = wishlistIds.has(product.id);
  const isOutOfStock = product.stock <= 0;
  const discount = product.discount ?? 0;
  const [qty, setQty] = useState(cartEntry?.qty ?? 0);

  const handleAdd = (e) => { e.stopPropagation(); addItem(product, 1); setQty((q) => q + 1); };
  const handleIncrement = (e) => { e.stopPropagation(); addItem(product, 1); setQty((q) => q + 1); };
  const handleDecrement = (e) => {
    e.stopPropagation();
    if (qty <= 1) { removeItem(product.id); setQty(0); } else { addItem(product, -1); setQty((q) => q - 1); }
  };
  const handleToggleWishlist = (e) => { e.stopPropagation(); toggleWishlist(product.id); };

  return (
    <motion.div
      className={`rrdevi-product-card${isOutOfStock ? ' is-out-of-stock' : ''}`}

      transition={{ duration: 0.15 }}
      onClick={() => navigate(`/products/${product.id}`)}
      style={{ cursor: 'pointer' }}
    >
      {discount > 0 && <span className="rrdevi-product-card__badge">{discount}% OFF</span>}

export default function NetworkStatus() {
  const [status, setStatus] = useState(null);
  const timeoutRef = useRef(null);

  useEffect(() => {
    const handleOffline = () => {
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
      setStatus('offline');
    };
    const handleOnline = () => {
      setStatus('online');
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
      timeoutRef.current = setTimeout(() => setStatus(null), 2500);
    };

    window.addEventListener('offline', handleOffline);
    window.addEventListener('online', handleOnline);
    return () => {
      window.removeEventListener('offline', handleOffline);
      window.removeEventListener('online', handleOnline);
      if (timeoutRef.current) clearTimeout(timeoutRef.current);
    };
  }, []);

  if (!status) return null;

  return (
    <div className={`rrdevi-network-status rrdevi-network-status--${status}`}>
      {status === 'offline' ? <WifiOff size={13} /> : <Wifi size={13} />}
      <span>{status === 'offline' ? "You're offline" : 'Back online'}</span>
    </div>
  );
}

    phone: user.phoneNumber ?? '',

    photoURL: user.photoURL ?? '',

  if (!order) return 0;
  if (order.deliveryStatus === 'DELIVERED') return 4;
  if (order.deliveryStatus === 'OUT_FOR_DELIVERY') return 3;
  if (order.deliveryStatus === 'PICKED') return 2;

import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { signInWithPhoneNumber, RecaptchaVerifier } from 'firebase/auth';
import { Truck, Phone } from 'lucide-react';
import { auth } from '../../firebase/config';

export default function DeliveryLogin() {
  const navigate = useNavigate();
  const [phone, setPhone] = useState('');
  const [otp, setOtp] = useState('');
  const [confirmationResult, setConfirmationResult] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const setupRecaptcha = () => {
    if (!window.recaptchaVerifier) {
      window.recaptchaVerifier = new RecaptchaVerifier(auth, 'delivery-recaptcha-container', { size: 'invisible' });
    }
    return window.recaptchaVerifier;
  };

  const handleSendOtp = async (e) => {
    e.preventDefault();
    if (!/^[6-9]\d{9}$/.test(phone)) { setError('Enter a valid 10-digit phone number.'); return; }
    setLoading(true);
    setError(null);
    try {
      const verifier = setupRecaptcha();
      const result = await signInWithPhoneNumber(auth, `+91${phone}`, verifier);
      setConfirmationResult(result);
    } catch (err) {
      console.error('OTP send failed:', err);
      setError('Could not send OTP. Please check the number and try again.');
    } finally {
      setLoading(false);
    }
  };

  const handleVerifyOtp = async (e) => {
    e.preventDefault();
    if (otp.length !== 6) { setError('Enter the 6-digit OTP.'); return; }
    setLoading(true);
    setError(null);
    try {
      await confirmationResult.confirm(otp);
      navigate('/delivery', { replace: true });
    } catch (err) {
      console.error('OTP verification failed:', err);
      setError('Invalid OTP. Please try again.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="rrdevi-delivery-login">
      <div className="rrdevi-delivery-login__brand">
        <Truck size={40} color="var(--color-primary)" />
        <h1>RR Devi Delivery Partner</h1>
      </div>

      {!confirmationResult ? (
        <form onSubmit={handleSendOtp} className="rrdevi-delivery-login__form">
          <label>Phone Number</label>
          <div className="rrdevi-delivery-login__phone-input">
            <span>+91</span>
            <input type="tel" maxLength={10} placeholder="9876543210" value={phone} onChange={(e) => setPhone(e.target.value.replace(/\D/g, ''))} />
          </div>
          {error && <p className="rrdevi-delivery-login__error">{error}</p>}
          <button type="submit" disabled={loading}><Phone size={16} /> {loading ? 'Sending OTP...' : 'Send OTP'}</button>
        </form>
      ) : (
        <form onSubmit={handleVerifyOtp} className="rrdevi-delivery-login__form">
          <label>Enter OTP sent to +91{phone}</label>
          <input type="text" maxLength={6} placeholder="6-digit OTP" value={otp} onChange={(e) => setOtp(e.target.value.replace(/\D/g, ''))} />
          {error && <p className="rrdevi-delivery-login__error">{error}</p>}
          <button type="submit" disabled={loading}>{loading ? 'Verifying...' : 'Verify & Login'}</button>
        </form>
      )}

      <div id="delivery-recaptcha-container" />
    </div>
  );
}

import { useState, useEffect } from 'react';
import { useParams } from 'react-router-dom';
import { doc, getDoc } from 'firebase/firestore';
import { Phone, MessageCircle, MapPin, Package, Navigation } from 'lucide-react';
import { db } from '../../firebase/config';
import { useDeliveryAuth } from '../../context/DeliveryAuthContext';
import { acceptOrder, advanceOrderStatus, confirmCodCollection } from '../../firebase/services/deliveryService';
import Skeleton from '../../components/common/Skeleton';

export default function DeliveryOrderDetails() {
  const { id } = useParams();
  const { deliveryProfile } = useDeliveryAuth();
  const [order, setOrder] = useState(null);
  const [loading, setLoading] = useState(true);
  const [actionLoading, setActionLoading] = useState(false);
  const [showCodPrompt, setShowCodPrompt] = useState(false);

  const fetchOrder = async () => {
    setLoading(true);
    const orderSnap = await getDoc(doc(db, 'orders', id));
    if (!orderSnap.exists()) { setOrder(null); setLoading(false); return; }
    const orderData = { id: orderSnap.id, ...orderSnap.data() };
    if (orderData.addressId) {
      const addressSnap = await getDoc(doc(db, 'addresses', orderData.addressId));
      if (addressSnap.exists()) orderData.resolvedAddress = addressSnap.data();
    }
    setOrder(orderData);
    setLoading(false);
  };

  useEffect(() => { fetchOrder(); }, [id]);

  const addressText = order?.resolvedAddress
    ? `${order.resolvedAddress.house}, ${order.resolvedAddress.street}, ${order.resolvedAddress.city} - ${order.resolvedAddress.pincode}`
    : 'Address unavailable';

  const handleAccept = async () => {
    setActionLoading(true);
    try { await acceptOrder(order.id, deliveryProfile.id); await fetchOrder(); }
    catch (err) { alert(err.message); }
    finally { setActionLoading(false); }
  };

  const handleAdvance = async () => {
    setActionLoading(true);
    try { await advanceOrderStatus(order.id, deliveryProfile.id, order.deliveryStatus); await fetchOrder(); }
    catch (err) { alert(err.message); }
    finally { setActionLoading(false); }
  };

  const handleMarkDelivered = () => {
    if (order.paymentMethod === 'COD') { setShowCodPrompt(true); return; }
    handleAdvance();
  };

  const handleCodConfirm = async (collected) => {
    setActionLoading(true);
    try {
      await advanceOrderStatus(order.id, deliveryProfile.id, order.deliveryStatus);
      await confirmCodCollection(order.id, collected, deliveryProfile.id);
      setShowCodPrompt(false);
      await fetchOrder();
    } catch (err) {
      alert(err.message);
    } finally {
      setActionLoading(false);
    }
  };

  const handleNavigate = () => {
    window.open(`https://www.google.com/maps/dir/?api=1&destination=${encodeURIComponent(addressText)}`, '_blank');
  };

  if (loading) return <div style={{ padding: 16 }}><Skeleton height="240px" radius="14px" /></div>;
  if (!order) return <div style={{ padding: 16 }}>Order not found.</div>;

  return (
    <div className="rrdevi-delivery-order-details">
      <h1>#{order.id.slice(-6).toUpperCase()}</h1>

      <section>
        <h2>Customer</h2>
        <p>{order.customerName}</p>
        <div className="rrdevi-delivery-order-details__contact-actions">
          <a href={`tel:${order.customerPhone}`}><Phone size={14} /> Call customer</a>
          <a href={`https://wa.me/91${order.customerPhone}`} target="_blank" rel="noopener noreferrer"><MessageCircle size={14} /> WhatsApp</a>
        </div>
      </section>

      <section>
        <h2><MapPin size={14} /> Delivery address</h2>
        <p>{addressText}</p>
      </section>

      <section>
        <h2><Package size={14} /> Items</h2>
        {order.items.map((item) => (
          <div key={item.id} className="rrdevi-delivery-order-details__item">
            <span>{item.name} × {item.qty}</span>
            <span>₹{item.price * item.qty}</span>
          </div>
        ))}
      </section>

      <section>
        <h2>Payment</h2>
        <p>{order.paymentMethod === 'COD' ? `Collect ₹${order.grandTotal} in cash` : 'Paid Online'}</p>
      </section>

      {order.deliveryStatus !== 'DELIVERED' && (
        <div className="rrdevi-delivery-order-details__actions">
          {order.deliveryStatus === 'ASSIGNED' && (
            <button disabled={actionLoading} onClick={handleAccept}>{actionLoading ? 'Updating...' : 'Accept Order'}</button>
          )}
          {order.deliveryStatus === 'ACCEPTED' && (
            <button disabled={actionLoading} onClick={handleAdvance}>{actionLoading ? 'Updating...' : 'Mark as Picked'}</button>
          )}
          {order.deliveryStatus === 'PICKED' && (
            <button disabled={actionLoading} onClick={handleAdvance}>{actionLoading ? 'Updating...' : 'Start Delivery'}</button>
          )}
          {order.deliveryStatus === 'OUT_FOR_DELIVERY' && (
            <>
              <button className="rrdevi-delivery-order-details__navigate" onClick={handleNavigate}><Navigation size={14} /> Navigate</button>
              <button disabled={actionLoading} onClick={handleMarkDelivered}>{actionLoading ? 'Updating...' : 'Mark Delivered'}</button>
            </>
          )}
        </div>
      )}

      {showCodPrompt && (
        <div className="rrdevi-delivery-order-details__cod-modal">
          <p>Payment collected?</p>
          <div>
            <button disabled={actionLoading} onClick={() => handleCodConfirm(true)}>Collected</button>
            <button disabled={actionLoading} onClick={() => handleCodConfirm(false)}>Not Collected</button>
          </div>
        </div>
      )}
    </div>
  );
}
